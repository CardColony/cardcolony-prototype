#### P1-A：插入整块（紧跟 `_projection_path` 的 `return` 之后，空两行）

```python
# ------------------------------------------------ Windows 文件锁竞态重试（2026-10-17）
# [WinErr13 竞态 · 2026-10-17 实测归纳] 重试预算：5 次尝试，指数退避
# 0.08/0.16/0.32/0.64 秒 ⇒ 单次写最多让出约 1.2 秒。
_LOCK_RETRY_ATTEMPTS = 5
_LOCK_RETRY_BASE_SLEEP = 0.08
_FILE_ATTRIBUTE_READONLY = 0x1


def _is_windows_lock_race(exc: OSError, path: Path) -> bool:
    """判「Windows 文件锁竞态」——只对可恢复的占用放行重试，其余原样上抛。

    Windows 上两类 errno 13（PermissionError）形态不同，必须分开：

    - ``winerror`` 5/32：**双路径**形态（``[WinError 5] 拒绝访问。: 'src' -> 'dst'``），
      来自 ``os.replace``／MoveFileEx 撞独占句柄——经典共享冲突，交给重试。
    - ``winerror`` 为 None 的**单路径**形态（``[Errno 13] Permission denied: 'path'``），
      来自 ``open()``——CRT 把 CreateFile 失败折成 errno 而不带 winerror。**这才是
      线上真实 502 的形态**：改动前唯独给 ``os.replace`` 配了重试，而 append/写 tmp 的
      ``open()`` 零重试，撞上即整轮 502。

    但它与「真无权限」（只读属性／ACL 拒绝）**异常形态完全相同**，仅凭异常分不开。
    故再查一次目标文件的只读属性：带只读位＝真无权限 ⇒ 立即上抛、零重试；
    不带只读位＝按锁竞态处理（有界重试）。
    """
    if os.name != "nt" or not isinstance(exc, PermissionError):
        return False
    winerror = getattr(exc, "winerror", None)
    if winerror in (5, 32):
        return True
    if winerror is not None:
        return False          # 其它 winerror（非本次两类）⇒ 行为不变，不重试
    try:
        attrs = os.stat(path).st_file_attributes
    except (OSError, AttributeError):
        return False          # 目标已消失等异常形态 ⇒ 不猜、不重试
    return not bool(attrs & _FILE_ATTRIBUTE_READONLY)


def _log_lock_race(exc: OSError, path: Path, attempt: int, what: str,
                   give_up: bool = False) -> None:
    """锁竞态对账日志：**每次写操作至多一条**（首次重试一条、重试耗尽再一条）。"""
    logger.warning(
        "[facade] Windows 文件锁竞态 %s %s：attempt=%d/%d winerror=%s errno=%s path=%s",
        "重试耗尽，上抛" if give_up else "退避重试",
        what, attempt + 1, _LOCK_RETRY_ATTEMPTS,
        getattr(exc, "winerror", None), getattr(exc, "errno", None), path)


def _open_with_lock_retry(path: Path, mode: str, what: str) -> Any:
    """按 mode 打开文件，遇 Windows 文件锁竞态做有界重试＋指数退避（真让出）。

    **只重试 ``open`` 本身**：失败时一个字节都还没落盘，重试无副作用；
    写盘／flush／fsync 一律留在调用方原地，其失败行为与改动前逐字相同。
    非锁竞态异常与重试耗尽 ⇒ 原对象上抛，不静默吞。
    """
    for attempt in range(_LOCK_RETRY_ATTEMPTS):
        try:
            return open(path, mode, encoding="utf-8")
        except PermissionError as exc:
            if attempt == _LOCK_RETRY_ATTEMPTS - 1 or not _is_windows_lock_race(exc, path):
                if attempt and _is_windows_lock_race(exc, path):
                    _log_lock_race(exc, path, attempt, what, give_up=True)
                raise
            if not attempt:
                _log_lock_race(exc, path, attempt, what)
            time.sleep(_LOCK_RETRY_BASE_SLEEP * (2 ** attempt))
    raise AssertionError("unreachable")  # pragma: no cover


def _replace_with_lock_retry(src: Path, dst: Path, what: str) -> None:
    """tmp→目标原子改名，带同一套有界重试＋退避。

    ``MoveFileEx`` 失败是原子的（要么换成功、要么源仍在），故重试不会产生半态；
    非「文件被占用」两类（如跨卷 EXDEV）照旧立即上抛。
    """
    for attempt in range(_LOCK_RETRY_ATTEMPTS):
        try:
            os.replace(src, dst)
            return
        except PermissionError as exc:
            if attempt == _LOCK_RETRY_ATTEMPTS - 1 or not _is_windows_lock_race(exc, dst):
                if attempt and _is_windows_lock_race(exc, dst):
                    _log_lock_race(exc, dst, attempt, what, give_up=True)
                raise
            if not attempt:
                _log_lock_race(exc, dst, attempt, what)
            time.sleep(_LOCK_RETRY_BASE_SLEEP * (2 ** attempt))
```

```python
# ---------------------------------------- Windows 文件锁竞态（快照侧）
# 判据与预算**全部复用** session_facade（P1 已落地），不另写第三份。
# 为何惰性 import：session_facade → card_store → data_snapshot 是同包真环，
# 模块级 import 会成环；本仓既有惯例同样是函数内 import。


def _race_path(exc: OSError, dst: Path) -> Path:
    """撞锁的是哪个文件——优先用异常自带的 ``filename``。

    实测：``copy2`` 撞**只读侧**的源（读被放行）不失败；真正被拒的是写目标
    那一步（errno=13, winerror=None）。但源若被 share=0 独占（连读都拒），
    失败点又回到源上。故不能想当然只查一个：异常指名了哪个就读哪个。
    """
    fn = getattr(exc, "filename", None)
    if isinstance(fn, str) and fn:
        try:
            return Path(fn)
        except (TypeError, ValueError):
            return dst
    return dst


def _copy_with_lock_retry(p: Path, dst: Path, wid: str) -> str | None:
    """单个文件进快照批次；撞 Windows 锁竞态做有界重试＋指数退避。

    返回 ``None``＝成功；返回错误串＝该文件失败（**不抛任何异常**）。
    非锁竞态（真无权限、ENOENT…）**零退避立即判失败**，不得让重试掩盖真错。
    """
    from .session_facade import (
        _LOCK_RETRY_ATTEMPTS, _LOCK_RETRY_BASE_SLEEP,
        _is_windows_lock_race, _log_lock_race,
    )

    what = f"快照拷贝 {wid}/{p.name}"
    for attempt in range(_LOCK_RETRY_ATTEMPTS):
        try:
            _copy_protected_file(p, dst)
            return None
        except PermissionError as exc:
            probe = _race_path(exc, dst)
            race = _is_windows_lock_race(exc, probe)
            if attempt == _LOCK_RETRY_ATTEMPTS - 1 or not race:
                if attempt and race:
                    _log_lock_race(exc, probe, attempt, what, give_up=True)
                _drop_partial(dst)
                return f"{type(exc).__name__}: {exc}"
            if not attempt:
                _log_lock_race(exc, probe, attempt, what)
            time.sleep(_LOCK_RETRY_BASE_SLEEP * (2 ** attempt))
        except (Exception, SystemExit) as exc:  # noqa: BLE001 SystemExit＝沙箱删除守卫
            # 非锁竞态（ENOENT／只读／IO…）：**零退避**立即判该文件失败。
            # 必须在这里收口：文件在列目录后、拷贝前消失（实测 cards.db-journal）
            # 会让一个文件的消失废掉整批——那正是本批要修的「整批作废」。
            _drop_partial(dst)
            return f"{type(exc).__name__}: {exc}"
    return "重试预算耗尽"  # pragma: no cover  (循环内必 return)


def _drop_partial(dst: Path) -> None:
    """失败后清掉可能写了一半的目标文件（尽力而为，绝不外抛）。

    为什么必须清：批次目录每次按时间戳新建，目标此前不存在，删除不碰历史好数据；
    而留着半截文件比缺文件更坏——恢复时会拿到一个静默损坏的副本。
    """
    try:
        if dst.exists():
            dst.unlink()
    except (Exception, SystemExit) as exc:  # noqa: BLE001 尽力而为
        logger.warning("半截快照文件清理失败（不影响对账）: %s (%s: %s)",
                       dst, type(exc).__name__, exc)
```

def _await_children_flag(meta: dict[str, Any]) -> bool:
    """读 ``goal_await_children`` 标志位（0/1；**机械读列，不猜**）。

    读侧容 ``1`` / ``"1"`` / ``True``：SQLite INTEGER 列经 JSON/前端往返可能成
    字符串或布尔。全归一成 bool，任一非空非零真值即为「在等子卡」。
    """
    raw = (meta or {}).get("goal_await_children")
    if raw is None or raw == "":
        return False
    if isinstance(raw, str):
        return raw.strip() not in ("", "0", "False", "false", "None")
    try:
        return int(raw) != 0
    except (TypeError, ValueError):
        return bool(raw)


def wake_verdict(work_id: str, card_id: str) -> str:
    """**向上唤醒专用**判据（纯判定，无副作用）——返回 :data:`WAKE_*` 三值之一。

    【为什么不用 ``will_continue`` 直接判】
    ``will_continue`` 仍**只认 active**，一个字不改——它被 ``routes._run_message``
    的回合末执行锁持有判定（``keep = will_continue(...)``）共用，改它会波及串行
    队列的锁语义。本判据**只**服务 ``after_turn`` 沿 parent 链的「唤醒」这一个动作。

    【缺口背景】父卡 ``update_goal(pause)`` 写「等子卡跑完回报」后恒为 paused；
    ``will_continue`` 判否 ⇒ ``after_turn`` 既不起它、也不再上跳 ⇒ 父卡**永久挂死**
    （实测 work_008：五张子卡全 complete，父卡 rounds 恒 0）。恢复看似"发生过"只是
    因为子卡**主动发了 msg_notify**——**模型行为掩盖了机制缺口**。

    判据＝``auto_mode=1`` ＋ 有 ``goal`` ＋ （``goal_state=="active"``
    **或**（``paused`` 且 ``goal_await_children==1``）） ＋ **无未终止直接子卡**。
    """
    meta = _meta(work_id, card_id)
    if not is_auto(meta):
        return WAKE_NONE
    if not str(meta.get("goal") or "").strip():
        return WAKE_NONE
    state = str(meta.get("goal_state") or "")
    if state == GOAL_ACTIVE:
        verdict = WAKE_ACTIVE
    elif state == GOAL_PAUSED and _await_children_flag(meta):
        verdict = WAKE_PAUSED
    else:
        return WAKE_NONE
    if has_active_child(work_id, card_id):
        return WAKE_NONE   # ★ 父卡让位（子卡还在跑 ⇒ 轮不到父卡）
    return verdict


async def _resume_awaiting(work_id: str, card_id: str) -> bool:
    """恢复「等子卡」挂起中的父卡：``goal_state → active`` 且**清**等待标志位。

    **一次 ``update_card``** 落两个字段（不留中间态）。清标志位的意义：它是**一次性
    意图**——「这次挂起是等子卡」用过即销，不跨目标生命周期留存。

    返回是否恢复成功（失败只留痕不抛——不得吃掉 ``after_turn`` 的整条链）。
    """
    try:
        await card_store.update_card(
            work_id, card_id,
            {"goal_state": GOAL_ACTIVE, "goal_await_children": 0},
            "[自动模式] 子卡完工，恢复等待中的父卡",
        )
    except Exception as exc:  # noqa: BLE001 恢复失败 ⇒ 不点火（否则状态自相矛盾）
        logger.warning("[auto] 等待中父卡恢复失败 work=%s card=%s: %s",
                       work_id, card_id, exc)
        return False
    return True
```

#: [6702 移植 2026-10-17] 宽裕度：未传参的纯推断 pause 在「子卡恰好刚完工」
#: 的采样边界上也可能是一次活锁重挂；对这类 pause 宽裕度地清零（把 1 次清掉）。
#: 取 1＝只抹平采样边界的一次歧义，不改变「连续 ≥3 次才死锁」的判定实效。
#: 它**不是闸**：不参与任何轮次/进度/时间/token 判定。

def _as_bool(raw: Any) -> bool:
    """宽松读布尔（模型可能送 ``true``/``"true"``/``1``/``"是"``）。

    缺省/空/``None`` ⇒ ``False``（**向后兼容的关键**：不传该参数时逐字保持旧行为）。
    刻意只把**明确真值**当真——宁可漏置位（退回旧推断分支）也不误置位。
    """
    if raw is None or raw == "":
        return False
    if isinstance(raw, bool):
        return raw
    if isinstance(raw, (int, float)):
        return raw != 0
    if isinstance(raw, str):
        return raw.strip().lower() in ("true", "1", "yes", "y", "是")
    return bool(raw)
```

#: 【它是什么】``goal_await_wakes``（**连续「空等重挂」次数**：模型声明
#: ``pause(await_children=true)`` 而**那一刻并无未终止子卡**——即等待条件早已兑现
#: 却仍把自己挂起——则 +1；声明时确有子卡在跑、或换新目标生命周期，则归零）
#: **达到本值**、且 wake 判定**此刻确无未终止子卡**时，``wake_verdict`` 的
#: ``WAKE_PAUSED`` 分支不再恢复（保持 paused），只留 ``logger.warning`` ＋
#: ``ops_log("auto_await_resume_giveup")`` 痕迹，等人处置。
#:
#: 【★ 计数点为什么在 pause 声明处，而不是恢复时】
#:   ① **对齐实测序列**：resume 时 +1 会让「第 1、2、3 次恢复都放行、第 4 次才拒」，
#:      而 work_017 的活锁形态与验收口径都是「第 3 次起不再恢复」。pause 点计数
#:      恰好第 3 次拒绝。
#:   ② **不误伤正常循环**：正常的「等→醒→干活→再等」循环里，每次 pause 声明时
#:      **确有子卡在跑** ⇒ 走清零分支，计数永远攒不起来；resume 时 +1 则会让一个
#:      完全合法的重复等待循环在第 4 次真等待时被永久卡死。
#:
#: 【★ 它为什么不是闸】
#:   1. **方向不同**：闸限制「能跑多少／花多少」；本值只回答「**这一次唤醒动作**
#:      要不要发生」。``wake_verdict`` 的返回值只被 ``after_turn`` 的「向上唤醒」
#:      分支读——**轮次续轮（active）根本不经过它**（那条路走 ``will_continue``）。
#:   2. **与进度无关**：不看轮次、不看墙钟、不看 token、不看卡片类型、不看完成度。
#:      它读的唯一输入是「**同一个等待条件已被满足却反复重挂**」。
#:   3. **不拦任何正常推进**：本卡若 active 走 WAKE_ACTIVE 分支，本值压根不被读；
#:      仍有未终止子卡时判据不成立（``has_active_child`` 先拦）；计数到多大都不拦
#:      「正常等待-被唤醒」。
#:   4. **它有出口**：兜底命中后卡保持 paused 且留 ops 痕迹；卡随时可由人恢复
#:      （见 P8 的 ``/auto/resume``）或由模型新的一次显式声明重新进入等待态。
#:   5. **与轮次闸对照**：真正的轮次闸会减少可跑的轮数；本值不减任何一轮。
#:
#: 【为什么是 3】主修复（唤醒时带明确事实的激励文）足以让绝大多数模型在第一、二次
#:   唤醒就整合收口；阈值取 3＝「给足两次修法生效的机会，第三次仍原地打转才认定
#:   是结构性死锁」。它**不是**「最多唤醒 3 次」：判定里带着「此刻无未终止子卡」，
#:   有子卡在跑时这个值再大也不生效。

#:
#: 【为什么用一次性标记，而不是给 ``kick`` 加参数】
#:   ``kick`` 是**对外**接口：``routes`` 与 ``kick_if_ready`` 都调它，给它加形参就把

```python
def _await_wakes(meta: dict[str, Any]) -> int:
    """读 ``goal_await_wakes`` 计数（机械读列，不猜；容 None/字符串/异常 → 0）。"""
    raw = (meta or {}).get("goal_await_wakes")
    if raw is None or raw == "":
        return 0
    try:
        return int(raw)
    except (TypeError, ValueError):
        return 0


def _giveup_await_resume(work_id: str, card_id: str, meta: dict[str, Any]) -> None:
    """死锁兜底命中时的**留痕**（不改任何状态，纯可见性）。

    卡保持 ``paused`` —— 本函数只打两条痕迹，把「为什么没被叫醒」落到日志与
    ops 档案里等人处置。**不改 goal_state、不改计数、不清 await 标志**。
    """
    wakes = _await_wakes(meta)
    reason = str(meta.get("goal") or "")[:200]
    logger.warning(
        "[auto] 等子卡唤醒死锁兜底：不再恢复 work=%s card=%s wakes=%d >= %d 且此刻无未终止子卡"
        "（模型恢复后重复挂起、来回打转；卡片保持 paused，请人工处置）",
        work_id, card_id, wakes, _AWAIT_RESUME_DEADLOCK_GIVEUP)
    try:
        session_ops.ops_log("auto_await_resume_giveup", work_id, card_id, {
            "goal_await_wakes": wakes,
            "threshold": _AWAIT_RESUME_DEADLOCK_GIVEUP,
            "has_active_child": False,
            "goal": reason,
            "action": "kept_paused",
        })
    except Exception as exc:  # noqa: BLE001 留痕失败不阻断
        logger.warning("[auto] 死锁兜底留痕失败 work=%s card=%s: %s", work_id, card_id, exc)


def _set_resume_note(note: str) -> None:
    """置位「本轮是等子卡恢复轮」的一次性标记。"""
    global _PENDING_RESUME_NOTE
    _PENDING_RESUME_NOTE = note


def _take_resume_note() -> str:
    """consume-once 取走唤醒事实标记（取走即清空，早退也不残留）。"""
    global _PENDING_RESUME_NOTE
    note = _PENDING_RESUME_NOTE or ""
    _PENDING_RESUME_NOTE = None
    return note

async def resume_card(work_id: str, card_id: str) -> bool:
    """**人工**恢复：``paused`` / ``blocked`` → ``active``，清等待位与活锁计数。

    与 ``_resume_awaiting`` 的区别：那个是**机制自动**恢复（只清 await 位、只从
    paused 恢复）；本函数是**人工**恢复——额外把 ``goal_await_wakes`` 归零（给人
    介入后的卡重新攒满 3 次机会），且允许从 ``blocked`` 恢复。

    为什么必须有它：死锁兜底（``_giveup_await_resume``）会把卡停在 paused，而全仓
    没有第二条路径能把 ``goal_state`` 写回 active（PATCH 白名单不含它、
    ``ensure_goal_from_message`` 在 goal 非空时直接返回、切 auto_mode 只重算 perms）。
    只留 ops 痕迹没有工具，不算出口。

    返回是否恢复成功（失败只留痕不抛）。
    """
    meta = _meta(work_id, card_id)
    if not is_auto(meta):
        return False
    if not str(meta.get("goal") or "").strip():
        return False
    try:
        await card_store.update_card(
            work_id, card_id,
            {"goal_state": GOAL_ACTIVE, "goal_await_children": 0,
             "goal_await_wakes": 0},
            "[自动模式] 人工恢复：重新推进目标",
        )
    except Exception as exc:  # noqa: BLE001 恢复失败 ⇒ 不起轮
        logger.warning("[auto] 人工恢复失败 work=%s card=%s: %s",
                       work_id, card_id, exc)
        return False
    kick(work_id, card_id)
    return True
```
