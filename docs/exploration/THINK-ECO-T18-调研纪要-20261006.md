## 二、C1 separateOverlaps 动作化（可照抄实现）

```js
// 输入：layout {cid:{x,y}}、sizes {cid:{w,h}}、pinnedIds:Set、GAP（默认 40）
function separateOverlaps(layout, sizes, pinnedIds, GAP = 40) {
  const pinned = pinnedIds || new Set();
  const out = {};
  for (const cid in layout) out[cid] = { x: layout[cid].x, y: layout[cid].y };
  const W = c => (sizes[cid] && sizes[cid].w) || 220;
  const H = c => (sizes[cid] && sizes[cid].h) || (depthCheck ? 36 : 86);

  // AABB 相交判定（含 GAP 扩张）
  const hits = (a, b) =>
    !(out[a].x + W(a) + GAP <= out[b].x || out[b].x + W(b) + GAP <= out[a].x ||
      out[a].y + H(a) + GAP <= out[b].y || out[b].y + H(b) + GAP <= out[a].y);

  // 迭代上限 16 轮（前端本地有限步 FPS 保护，非模型调用闸——辨析见任务书红线）
  for (let round = 0; round < 16; round++) {
    let moved = false;
    const ids = Object.keys(layout);
    for (let i = 0; i < ids.length; i++) {
      for (let j = i + 1; j < ids.length; j++) {
        const a = ids[i], b = ids[j];
        if (!hits(a, b)) continue;
        // MTD：重叠量（含 GAP）
        const ox = (out[a].x + W(a) + GAP) - out[b].x;      // a 在 b 左时的横向重叠
        const oy = (out[a].y + H(a) + GAP) - out[b].y;      // a 在 b 上时的纵向重叠
        // 纵向优先：|oy| <= |ox| 推 y，否则推 x（保树形列结构）
        const pushY = Math.abs(oy) <= Math.abs(ox);
        if (pinned.has(a) && pinned.has(b)) continue;       // 双 pinned：不动＋warn（C3 不新增，历史由整理画布清）
        const moveBy = (cid, dx, dy) => {
          if (pinned.has(cid)) return;                       // pinned 永不被移动
          out[cid].x += dx; out[cid].y += dy; moved = true;
        };
        if (pushY) {
          const aBelow = out[a].y <= out[b].y;
          if (aBelow) { moveBy(a, 0, oy); moveBy(b, 0, -oy * (pinned.has(b) ? 1 : 0.5)); }
          else       { moveBy(b, 0, oy * (pinned.has(a) ? 0.5 : 1)); moveBy(b, 0, oy); }
          // free-pinned 推 free 全量；free-free 各推半量——上半伪代码须在 taskbook 里按
          // 「pinned 永不动」拆成明确分支（见下方第三条）
        } else {
          // x 向对称处理（同上，把 y 换 x）
        }
      }
    }
    if (!moved) break;   // 本轮无移动＝已收敛
  }
  // 不收敛截断＋warn（如实标注可能残留）
  console.warn("[separateOverlaps] 16 轮内未完全收敛，可能残留个别相交");
  return out;
}
```

**taskbook 必须把上方伪代码中 `moveBy` 的 pinned 分支拆成三条明确规则（写死）：**
1. **pinned 永不被移动**——任何时候 `pinned.has(cid)` 则该卡坐标不变。
2. **free–pinned 相交**：推 free 的 MTD 全量（pinned 不动）。
3. **free–free 相交**：两卡各推 MTD 半量。
4. 双 pinned 相交：不动＋warn（不产生移动；由 C3 保证不再新增、历史遗留交「整理画布」按钮）。

**GAP 引用**：separateOverlaps 用默认参数 `GAP=40`；且 `reflowLayout`（C2）调用处传 `GAP_Y` 常量（L310/L322 已有）——消除 L380 那处硬编码 40（taskbook 收口项③：正式服 resolveOverlaps L380 `prevBottom + 40` 改为引用 GAP_Y；测试空间 L396 同）。

## 三、C3 resolveDrop 动作化（可照抄实现）

```js
// 松手点仲裁：onDragUp moved 分支内、persistCardPos 之前
function resolveDrop(cid, x, y) {
  const GAP = 40;
  const others = 当前画布可见集中除 cid 外的所有卡（用 state._baseLayout/_layout＋measureSizes() 取矩形）；
  let closest = null, bestShift = Infinity;
  for (const [oid, o] of others) {
    if (aabb 无相交(x,y,220,86, o)) continue;   // 无相交 = 零改动路径
    // 方案一：y 向最近空档——同列卡按 y 排序，找 ≥卡高+GAP 的间隙
    const gapY = findVerticalGap(oid, 卡高+GAP);   // 找该卡下方≥GAP 的最近空位
    const shiftY = Math.abs(gapY - y);
    if (shiftY < bestShift) { bestShift = shiftY; closest = { x, y: gapY }; }
    // 方案二：x 向推出
    const shiftX = 使不重叠的最小横向位移;
    if (shiftX < bestShift) { bestShift = shiftX; closest = { x: shiftX, y }; }
  }
  if (!closest) return { x, y };      // 无相交，零改动
  return closest;                     // 取位移最小者
}
// 交回 persistCardPos(cid, closest.x, closest.y)，pinned:true 语义不变
```

**taskbook 把三处写死：** ①「无相交→零改动路径」是 C3 核心（拖到空地原样落）；②有相交→最小位移到不交位（y 向最近空档优先、其次 x 向推出、取位移最小）；③正在拖的卡让先在场的卡（P1/P2 一致）。
